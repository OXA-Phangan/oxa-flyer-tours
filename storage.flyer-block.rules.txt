    // ---- OXA Flyer Tours (separate app, same Firebase project) ----
    // Duplicated locally since Storage rules can't reference Firestore
    // rules functions — same pattern as isPosterManagement()/
    // isPortalManagement() above. Keep in sync with firestore.rules'
    // isFlyerAdmin() if the claim name ever changes.
    function isFlyerAdmin() {
      return isSignedIn() &&
             (request.auth.token.flyerAdmin == true ||
              request.auth.token.email.lower() in
                ['info@oxaphangan.com', 'bella.oxaphangan@gmail.com', 'mobit.booking@gmail.com']);
    }

    // Passport photos — path shape: flyerPassports/{token}/passport.jpg
    // get per known token (exact path) is allowed, list (directory
    // enumeration) is admin-only — same "token = access, no browsing"
    // model as the Firestore rules.
    match /flyerPassports/{token}/{fileName} {
      allow get: if true;
      allow list: if isFlyerAdmin();
      allow write: if true;
      allow delete: if isFlyerAdmin();
    }

    // Tour proof photos — path shape: flyerProofs/{token}/{proofId}.jpg
    match /flyerProofs/{token}/{fileName} {
      allow get: if true;
      allow list: if isFlyerAdmin();
      allow write: if true;
      allow delete: if isFlyerAdmin();
    }

    // Chat media (photos, videos, voice messages) — path shapes:
    //   flyerChat/{volunteerToken}/{file}
    //   flyerChat/group/{volunteerToken | admin}/{file}
    // Same "token = access, no browsing" model; create-only (no overwrite),
    // size- and type-limited. Deleted by cleanupExpiredFlyerChatMedia.
    match /flyerChat/{allPaths=**} {
      allow get: if true;
      allow list: if isFlyerAdmin();
      allow create: if request.resource.size < 30 * 1024 * 1024
                    && request.resource.contentType.matches('(image|video|audio)/.*');
      allow update, delete: if isFlyerAdmin();
    }

    // Bike rental condition photos — path shape: flyerBike/{token}/{file}.jpg
    // Taken live in the volunteer portal before a rental. Same "token = access,
    // no browsing" model; create-only, images up to 10 MB. Deleted 7 days after
    // check-out by cleanupExpiredFlyerBikePhotos.
    match /flyerBike/{token}/{fileName} {
      allow get: if true;
      allow list: if isFlyerAdmin();
      allow create: if request.resource.size < 10 * 1024 * 1024
                    && request.resource.contentType.matches('image/.*');
      allow update, delete: if isFlyerAdmin();
    }
