    // ---- OXA Flyer Tours (separate app, same Firebase project) ----
    // See github.com/OXA-Phangan/oxa-flyer-tours. Short-term flyer
    // volunteers (max ~2 weeks), never entered into `employees` — access
    // is a per-volunteer, unique-link token (flyerVolunteers/{token}),
    // not Google Sign-In like every other app above. Admin access uses
    // a distinct `flyerAdmin` custom claim (isFlyerAdmin() below) rather
    // than the shared isAdmin()/isPortalManagement() helpers above, since
    // this is a deliberately separate access domain.
    function isFlyerAdmin() {
      return isSignedIn() &&
             (request.auth.token.flyerAdmin == true ||
              request.auth.token.email.lower() in
                ['info@oxaphangan.com', 'bella.oxaphangan@gmail.com', 'mobit.booking@gmail.com']);
    }

    match /flyerRegistrations/{regId} {
      allow create: if request.resource.data.status == "pending"
                    && request.resource.data.name is string
                    && request.resource.data.checkInDate is string
                    && request.resource.data.checkOutDate is string;
      allow read, update, delete: if isFlyerAdmin();
    }

    // Collection-group read of every volunteer's shifts (employee portal
    // calendar). A per-path rule does NOT cover collectionGroup() queries.
    match /{path=**}/shifts/{shiftId} {
      allow read: if isFlyerAdmin() || isPortalManagement();
    }

    // One doc per bike per day: flyerBikeSlots/{YYYY-MM-DD}_{red|blue}. Volunteers can
    // only CREATE it (never overwrite), so a bike can't be requested twice for the same
    // day — even when two volunteers submit at the same moment. Admin cancels = delete.
    match /flyerBikeSlots/{slotId} {
      allow get: if true;
      allow list: if isFlyerAdmin() || isPortalManagement();
      allow create: if isFlyerAdmin() || (
        request.resource.data.keys().hasOnly(['day', 'bike', 'volunteerId', 'status', 'createdAt']) &&
        request.resource.data.bike in ['red', 'blue'] &&
        request.resource.data.day.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$') &&
        slotId == request.resource.data.day + '_' + request.resource.data.bike &&
        request.resource.data.volunteerId is string &&
        exists(/databases/$(database)/documents/flyerVolunteers/$(request.resource.data.volunteerId)) &&
        request.resource.data.status == 'requested' &&
        request.resource.data.createdAt == request.time
      );
      allow update, delete: if isFlyerAdmin();
    }

    // Admin "Bikes" tab + employee portal: all rentals across volunteers (read-only for management).
    // (Top level of the rules file's `match /databases/{database}/documents` — like the existing
    //  `match /{path=**}/shifts/{shiftId}` rule.)
    match /{path=**}/bikeRentals/{rentalId} {
      allow read: if isFlyerAdmin() || isPortalManagement();
    }

    match /flyerRoutes/{routeId} {
      allow get, list: if true;
      allow create, update, delete: if isFlyerAdmin();
    }

    match /flyerVolunteers/{token} {
      allow get: if true;
      // Management (employee portal calendar) may list too — read-only.
      allow list: if isFlyerAdmin() || isPortalManagement();
      allow create: if isFlyerAdmin();
      allow update: if isFlyerAdmin() || (
        request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(["pendingStayEdit", "hasPendingStayEdit", "onBreak"])
      );
      allow delete: if isFlyerAdmin();

      match /shifts/{shiftId} {
        allow get, list: if true;
        allow write: if isFlyerAdmin();
      }

      // Per-volunteer rentals / extra charges, same "token = access" model as shifts:
      //   flyerVolunteers/{token}/bikeRentals/{id}
      // A volunteer may only CREATE a plain "requested" rental for the standard price
      // (photoPaths 1..8, createdAt == request.time). Everything else is admin-only.
      // (Put this inside the existing `match /flyerVolunteers/{token} { ... }` block,
      //  next to `match /shifts/{shiftId}`.)
      match /bikeRentals/{rentalId} {
        allow get, list: if true;
        allow create: if isFlyerAdmin() || (
          request.resource.data.keys().hasOnly(['kind', 'volunteerId', 'volunteerName', 'day', 'bike', 'status',
                                                'amount', 'issues', 'note', 'photoPaths', 'createdBy', 'createdAt']) &&
          request.resource.data.kind == 'rental' &&
          request.resource.data.volunteerId == token &&
          request.resource.data.volunteerName is string &&
          request.resource.data.day is string &&
          request.resource.data.day.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$') &&
          request.resource.data.bike in ['red', 'blue'] &&
          request.resource.data.status == 'requested' &&
          request.resource.data.amount == 150 &&
          request.resource.data.createdBy == 'volunteer' &&
          request.resource.data.photoPaths is list &&
          request.resource.data.photoPaths.size() >= 1 &&
          request.resource.data.photoPaths.size() <= 8 &&
          request.resource.data.createdAt == request.time
        );
        allow update, delete: if isFlyerAdmin();
      }
    }

    match /flyerTourProgress/{progressId} {
      allow get, write: if true;
      // Management: employee portal "Volunteers / Flyer Tours" tile (read-only).
      allow list: if isFlyerAdmin() || isPortalManagement();
    }

    // Append-only tour event log (tour_start / spot_completed / spot_skipped /
    // spot_passed / break_start / break_end / tour_finished), written by the
    // volunteer portal (unauthenticated token link, like every other flyer
    // write) and read by the employee portal's Flyer Tour timeline. Never
    // updated or deleted. `at` must be the server timestamp.
    match /flyerTourEvents/{eventId} {
      allow create: if request.resource.data.keys().hasOnly(['volunteerId', 'volunteerName', 'day', 'region', 'routeId', 'routeName',
                                                              'type', 'spotIndex', 'spotName', 'spotType', 'reason', 'proofStoragePath', 'at']) &&
                    request.resource.data.volunteerId is string &&
                    exists(/databases/$(database)/documents/flyerVolunteers/$(request.resource.data.volunteerId)) &&
                    request.resource.data.volunteerName is string &&
                    request.resource.data.day is string &&
                    request.resource.data.day.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$') &&
                    request.resource.data.type in ['tour_start', 'spot_completed', 'spot_skipped', 'spot_passed',
                                                   'break_start', 'break_end', 'tour_finished'] &&
                    request.resource.data.at == request.time;
      allow read: if isFlyerAdmin() || isPortalManagement();
      allow update, delete: if false;
    }

    // Small admin-side state shared between the flyer admin page and the
    // employee portal (e.g. groupChat.seenAt for the group-chat unread dot,
    // which the admin page otherwise only kept in its own localStorage).
    match /flyerAdminState/{docId} {
      allow read, write: if isFlyerAdmin() || isPortalManagement();
    }

    // Editable texts of the public registration form (Terms & Conditions, confirmation
    // message): flyerSettings/registration. Public read (the unauthenticated /register
    // page needs it), admin-only write. Top level of the Flyer section.
    match /flyerSettings/{docId} {
      allow get: if true;
      allow list: if isFlyerAdmin() || isPortalManagement();
      allow write: if isFlyerAdmin();
    }

    match /flyerProofs/{proofId} {
      allow create: if request.resource.data.volunteerId is string;
      allow get: if true;
      // Management may list (timeline proof photos), not delete.
      allow list: if isFlyerAdmin() || isPortalManagement();
      allow delete: if isFlyerAdmin();
    }

    match /flyerChatThreads/{token} {
      allow get, update: if true;
      // Management may list (Chat button unread dot on the portal tile).
      allow list: if isFlyerAdmin() || isPortalManagement();
      allow create: if isFlyerAdmin();
      match /messages/{messageId} {
        allow get, list, create: if true;
      }
    }

    match /flyerGroupMessages/{messageId} {
      allow get, list, create: if true;
      allow update, delete: if isFlyerAdmin();
    }

    match /flyerSupplyReports/{reportId} {
      allow create: if request.resource.data.volunteerId is string;
      allow get, list, update: if isFlyerAdmin();
    }
