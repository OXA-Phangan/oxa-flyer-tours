
// =====================================================================
// OXA Flyer Tours — chat media retention (separate app, same Firebase project)
// Source: github.com/OXA-Phangan/oxa-flyer-tours (cloud-functions.flyer-chat-media-cleanup.js.txt)
//
// Photos, videos and voice messages sent in the flyer chats live in Storage:
//   flyerChat/{volunteerToken}/{file}         direct chat with that volunteer
//   flyerChat/group/{volunteerToken}/{file}   group chat, sent by that volunteer
//   flyerChat/group/admin/{file}              group chat, sent by OXA
// Files are deleted FLYER_PASSPORT_GRACE_DAYS (7) after the (sender) volunteer's
// check-out — i.e. at flyerVolunteers/{token}.passportDeleteAfter, which already
// follows stay changes. OXA's group uploads have no stay, so they go 7 days after
// upload. Files whose volunteer no longer exists: 7 days after upload. Hard cap
// for everything else: 90 days. The chat message docs stay; the app shows
// "Media deleted" once the file is gone. Needs the helpers/constants of
// cloud-functions.flyer-photo-cleanup.js.txt (flyerDeleteFile, FLYER_HOUR_MS, ...).
// Runs daily (Admin SDK, bypasses storage.rules). Every deletion is logged.
// =====================================================================
exports.cleanupExpiredFlyerChatMedia = onSchedule(
  { schedule: "every day 03:30", timeZone: "Asia/Bangkok" },
  async () => {
    const db = getFirestore();
    const bucket = getStorage().bucket();
    const now = Date.now();
    const dayMs = 24 * FLYER_HOUR_MS;
    const volunteerDeadline = new Map(); // token -> ms | null (no volunteer doc)

    const [files] = await bucket.getFiles({ prefix: "flyerChat/" });
    for (const file of files) {
      const createdMs = Date.parse(file.metadata && file.metadata.timeCreated);
      if (!Number.isFinite(createdMs)) continue;

      const parts = file.name.split("/"); // flyerChat/<token>/<file> | flyerChat/group/<token|admin>/<file>
      const token = parts[1] === "group" ? (parts.length >= 4 ? parts[2] : null) : parts.length >= 3 ? parts[1] : null;

      let deadline = createdMs + FLYER_PASSPORT_GRACE_DAYS * dayMs; // OXA group uploads / unknown volunteer
      if (token && token !== "admin") {
        if (!volunteerDeadline.has(token)) {
          const snap = await db.collection("flyerVolunteers").doc(token).get();
          let ms = null;
          if (snap.exists) {
            const after = snap.get("passportDeleteAfter");
            ms = after && typeof after.toMillis === "function" ? after.toMillis() : null;
          }
          volunteerDeadline.set(token, ms);
        }
        const ms = volunteerDeadline.get(token);
        if (ms !== null) deadline = ms;
      }
      deadline = Math.min(deadline, createdMs + 90 * dayMs);

      if (deadline <= now) await flyerDeleteFile(bucket, file.name);
    }
  }
);
